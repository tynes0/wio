cmake_minimum_required(VERSION 3.20)
project(vm_sdk_task_inputs_host LANGUAGES CXX)
find_package(WioVmRuntime CONFIG REQUIRED)
wio_add_vm_module(vm_task_inputs_shared BYTECODE "${WIO_TEST_BYTECODE}")
wio_add_vm_module(vm_task_inputs_static STATIC BYTECODE "${WIO_TEST_BYTECODE}" GETTER WioStaticTaskInputsContract)
wio_add_vm_module(vm_task_inputs_unsupported BYTECODE "${WIO_TEST_BAD_BYTECODE}")
add_executable(vm_task_inputs_host "${WIO_TEST_HOST_SOURCE}")
target_compile_features(vm_task_inputs_host PRIVATE cxx_std_20)
target_link_libraries(vm_task_inputs_host PRIVATE vm_task_inputs_static Wio::wio_sdk ${CMAKE_DL_LIBS} Threads::Threads)
if(MSVC AND WioVmRuntime_MSVC_STATIC_RUNTIME)
    set_property(TARGET vm_task_inputs_host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_dependencies(vm_task_inputs_host vm_task_inputs_shared vm_task_inputs_unsupported)
enable_testing()
add_test(NAME installed_vm_task_inputs_host COMMAND vm_task_inputs_host
    "$<TARGET_FILE:vm_task_inputs_shared>" "$<TARGET_FILE:vm_task_inputs_unsupported>")
set_tests_properties(installed_vm_task_inputs_host PROPERTIES TIMEOUT 45)
