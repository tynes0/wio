cmake_minimum_required(VERSION 3.20)
project(vm_sdk_module_host LANGUAGES CXX)

find_package(WioVmRuntime CONFIG REQUIRED)
wio_add_vm_module(vm_shared BYTECODE "${WIO_TEST_BYTECODE}")
wio_add_vm_module(vm_static STATIC BYTECODE "${WIO_TEST_BYTECODE}" GETTER WioStaticTestModule)
wio_add_vm_module(vm_second STATIC BYTECODE "${WIO_TEST_BYTECODE}" GETTER WioStaticSecondModule)
wio_add_vm_module(vm_invalid BYTECODE "${WIO_TEST_INVALID_BYTECODE}")
add_executable(vm_sdk_host "${WIO_TEST_HOST_SOURCE}")
target_compile_features(vm_sdk_host PRIVATE cxx_std_20)
target_link_libraries(vm_sdk_host PRIVATE vm_static vm_second Wio::wio_sdk ${CMAKE_DL_LIBS} Threads::Threads)
if(MSVC AND WioVmRuntime_MSVC_STATIC_RUNTIME)
    set_property(TARGET vm_sdk_host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_custom_command(TARGET vm_sdk_host POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:vm_shared>" "$<TARGET_FILE_DIR:vm_sdk_host>/deployed-vm${CMAKE_SHARED_LIBRARY_SUFFIX}")
add_dependencies(vm_sdk_host vm_shared)
add_executable(vm_invalid_host "${WIO_TEST_INVALID_HOST_SOURCE}")
target_compile_features(vm_invalid_host PRIVATE cxx_std_20)
target_link_libraries(vm_invalid_host PRIVATE Wio::wio_sdk ${CMAKE_DL_LIBS})
add_dependencies(vm_invalid_host vm_invalid)
enable_testing()
add_test(NAME installed_sdk_host COMMAND vm_sdk_host "$<TARGET_FILE_DIR:vm_sdk_host>/deployed-vm${CMAKE_SHARED_LIBRARY_SUFFIX}")
set_tests_properties(installed_sdk_host PROPERTIES TIMEOUT 30)
add_test(NAME invalid_installed_sdk_host COMMAND vm_invalid_host "$<TARGET_FILE:vm_invalid>")
set_tests_properties(invalid_installed_sdk_host PROPERTIES TIMEOUT 30)
