cmake_minimum_required(VERSION 3.20)
project(vm_sdk_async_host LANGUAGES CXX)
find_package(WioVmRuntime CONFIG REQUIRED)
wio_add_vm_module(vm_async_shared BYTECODE "${WIO_TEST_BYTECODE}")
wio_add_vm_module(vm_async_static STATIC BYTECODE "${WIO_TEST_BYTECODE}" GETTER WioStaticAsyncContract)
add_executable(vm_async_host "${WIO_TEST_HOST_SOURCE}")
target_compile_features(vm_async_host PRIVATE cxx_std_20)
target_link_libraries(vm_async_host PRIVATE vm_async_static Wio::wio_sdk ${CMAKE_DL_LIBS} Threads::Threads)
if(MSVC AND WioVmRuntime_MSVC_STATIC_RUNTIME)
    set_property(TARGET vm_async_host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_dependencies(vm_async_host vm_async_shared)
enable_testing()
add_test(NAME installed_vm_async_host COMMAND vm_async_host "$<TARGET_FILE:vm_async_shared>")
set_tests_properties(installed_vm_async_host PROPERTIES TIMEOUT 45)
