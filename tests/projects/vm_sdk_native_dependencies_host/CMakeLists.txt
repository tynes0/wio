cmake_minimum_required(VERSION 3.20)
project(vm_sdk_native_dependencies_host LANGUAGES CXX)
find_package(WioVmRuntime CONFIG REQUIRED)
if(NOT DEFINED WIO_DEPLOY_BIAS)
    set(WIO_DEPLOY_BIAS 100)
endif()
add_library(deploy_calc SHARED "${WIO_TEST_FIXTURES}/vm_deployment_calc.cpp")
target_compile_definitions(deploy_calc PRIVATE WIO_CALC_BUILD "WIO_DEPLOY_BIAS=${WIO_DEPLOY_BIAS}")
add_library(deploy_native SHARED "${WIO_TEST_FIXTURES}/vm_deployment_producer.cpp")
target_link_libraries(deploy_native PRIVATE deploy_calc Wio::wio_sdk)
add_library(deploy_invalid SHARED "${WIO_TEST_FIXTURES}/vm_deployment_invalid.cpp")
target_link_libraries(deploy_invalid PRIVATE Wio::wio_sdk)
foreach(target IN ITEMS deploy_calc deploy_native deploy_invalid)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/original-producers"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/original-producers" BUILD_RPATH_USE_ORIGIN TRUE)
endforeach()
if(APPLE)
    # BUILD_RPATH_USE_ORIGIN is ELF-specific. Preserve sibling dependency lookup
    # after copying Mach-O producers into an immutable native cohort as well.
    set_target_properties(deploy_calc deploy_native deploy_invalid PROPERTIES BUILD_RPATH "@loader_path")
endif()
wio_add_vm_module(native_shared BYTECODE "${WIO_TEST_BYTECODE}"
    NATIVE_LIBRARIES deploy_native RUNTIME_LIBRARIES deploy_calc)
wio_add_vm_module(native_static STATIC BYTECODE "${WIO_TEST_BYTECODE}" GETTER WioStaticNativeContract
    NATIVE_LIBRARIES deploy_native RUNTIME_LIBRARIES deploy_calc)
wio_add_vm_module(native_bad BYTECODE "${WIO_TEST_BYTECODE}"
    NATIVE_LIBRARIES deploy_native deploy_invalid RUNTIME_LIBRARIES deploy_calc)
add_executable(native_host "${WIO_TEST_HOST_SOURCE}")
target_link_libraries(native_host PRIVATE native_static Wio::wio_sdk ${CMAKE_DL_LIBS} Threads::Threads)
target_compile_definitions(native_host PRIVATE "WIO_DEPLOY_BIAS=${WIO_DEPLOY_BIAS}")
if(MINGW)
    target_link_options(native_host PRIVATE -municode)
endif()
wio_deploy_vm_module(native_static HOST native_host)
add_dependencies(native_host native_shared native_bad)
set_target_properties(native_host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/final-host")
if(WIO_TEST_INSTALL_WORKFLOW)
    if(WIO_TEST_INSTALL_POLICY STREQUAL "traversal")
        wio_install_vm_module(native_shared DESTINATION "../escape")
    elseif(WIO_TEST_INSTALL_POLICY STREQUAL "absolute")
        wio_install_vm_module(native_shared DESTINATION "${CMAKE_BINARY_DIR}/escape")
    elseif(WIO_TEST_INSTALL_POLICY STREQUAL "static_without_host")
        wio_install_vm_module(native_static DESTINATION bin)
    elseif(WIO_TEST_INSTALL_POLICY STREQUAL "shared_with_host")
        wio_install_vm_module(native_shared HOST native_host DESTINATION bin)
    elseif(WIO_TEST_INSTALL_POLICY STREQUAL "not_vm")
        wio_install_vm_module(deploy_calc DESTINATION bin)
    elseif(WIO_TEST_INSTALL_POLICY STREQUAL "archive_host")
        wio_install_vm_module(native_static HOST native_static DESTINATION bin)
    elseif(WIO_TEST_INSTALL_POLICY STREQUAL "duplicate_repair")
        wio_deploy_vm_module(native_shared HOST native_shared DEPLOY_TARGET native_host)
    elseif(WIO_TEST_INSTALL_POLICY)
        message(FATAL_ERROR "Unknown VM install policy fixture")
    endif()
    foreach(target IN ITEMS native_shared native_bad)
        wio_deploy_vm_module(${target} HOST ${target} DEPLOY_TARGET repair_${target})
        wio_install_vm_module(${target} DESTINATION bin COMPONENT VmFixture)
    endforeach()
    wio_deploy_vm_module(native_static HOST native_host DEPLOY_TARGET repair_native_static)
    wio_install_vm_module(native_static HOST native_host DESTINATION bin COMPONENT VmFixture)
endif()
if(MSVC AND WioVmRuntime_MSVC_STATIC_RUNTIME)
    foreach(target IN ITEMS deploy_calc deploy_native deploy_invalid native_host)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
endif()
get_target_property(shared_manifest native_shared WIO_VM_NATIVE_MANIFEST)
get_target_property(static_manifest native_static WIO_VM_NATIVE_MANIFEST)
get_target_property(bad_manifest native_bad WIO_VM_NATIVE_MANIFEST)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/paths-$<CONFIG>.cmake" CONTENT
"set(test_shared [==[$<TARGET_FILE:native_shared>]==])
set(test_bad [==[$<TARGET_FILE:native_bad>]==])
set(test_host [==[$<TARGET_FILE:native_host>]==])
set(test_shared_manifest [==[${shared_manifest}]==])
set(test_static_manifest [==[${static_manifest}]==])
set(test_bad_manifest [==[${bad_manifest}]==])
set(test_producer_name [==[$<TARGET_FILE_NAME:deploy_native>]==])
set(test_runtime_name [==[$<TARGET_FILE_NAME:deploy_calc>]==])
")
