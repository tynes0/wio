cmake_minimum_required(VERSION 3.20)
project(vm_sdk_session_host LANGUAGES CXX)
find_package(WioVmRuntime CONFIG REQUIRED)
wio_add_vm_module(vm_session_shared BYTECODE "${WIO_TEST_BYTECODE}")
wio_add_vm_module(vm_session_static STATIC BYTECODE "${WIO_TEST_BYTECODE}" GETTER WioStaticSessionContract)
add_executable(vm_session_host "${WIO_TEST_HOST_SOURCE}")
target_compile_features(vm_session_host PRIVATE cxx_std_20)
target_link_libraries(vm_session_host PRIVATE vm_session_static Wio::wio_sdk ${CMAKE_DL_LIBS} Threads::Threads)
if(MSVC AND WioVmRuntime_MSVC_STATIC_RUNTIME)
    set_property(TARGET vm_session_host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_dependencies(vm_session_host vm_session_shared)
enable_testing()
add_test(NAME installed_vm_session_host COMMAND vm_session_host "$<TARGET_FILE:vm_session_shared>")
set_tests_properties(installed_vm_session_host PROPERTIES TIMEOUT 30)
