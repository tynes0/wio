cmake_minimum_required(VERSION 3.20)
project(WioVmProject LANGUAGES CXX)
find_package(WioVmRuntime CONFIG REQUIRED)
file(READ "${WIO_VM_PLAN}" plan)
function(read_scalar key)
    string(JSON value GET "${plan}" "${key}")
    if(value MATCHES ";")
        message(FATAL_ERROR "VM project paths cannot contain CMake list separators")
    endif()
    set(${key} "${value}" PARENT_SCOPE)
endfunction()
function(read_array key)
    string(JSON count LENGTH "${plan}" "${key}")
    set(values)
    if(count GREATER 0)
        math(EXPR last "${count} - 1")
        foreach(index RANGE 0 ${last})
            string(JSON value GET "${plan}" "${key}" ${index})
            if(value MATCHES ";")
                message(FATAL_ERROR "VM project paths/arguments cannot contain CMake list separators")
            endif()
            list(APPEND values "${value}")
        endforeach()
    endif()
    set(${key} "${values}" PARENT_SCOPE)
endfunction()
foreach(key IN ITEMS kind bytecode module host getter)
    read_scalar("${key}")
endforeach()
foreach(key IN ITEMS producers runtimeLibraries hostSources hostIncludes hostLinkDirs hostLibraries hostArguments hostLinkArguments)
    read_array("${key}")
endforeach()
if(NOT kind MATCHES "^(shared|static)$" OR NOT IS_ABSOLUTE "${module}" OR
   (host AND NOT IS_ABSOLUTE "${host}"))
    message(FATAL_ERROR "Invalid VM project target/output plan")
endif()
if(CMAKE_BUILD_TYPE AND NOT CMAKE_BUILD_TYPE MATCHES "^[A-Za-z0-9_+-]+$")
    message(FATAL_ERROR "VM project configuration must be a portable configuration name")
endif()
get_filename_component(package_dir "${CMAKE_CURRENT_LIST_DIR}" DIRECTORY)
include("${package_dir}/WioVmNativePaths.cmake")
function(output_path target output)
    get_filename_component(directory "${output}" DIRECTORY)
    get_filename_component(name "${output}" NAME_WLE)
    get_filename_component(suffix "${output}" LAST_EXT)
    set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${name}" SUFFIX "${suffix}"
        RUNTIME_OUTPUT_DIRECTORY "${directory}/$<0:>" LIBRARY_OUTPUT_DIRECTORY "${directory}/$<0:>"
        ARCHIVE_OUTPUT_DIRECTORY "${directory}/$<0:>")
endfunction()
string(TOUPPER "${kind}" module_kind)
if(kind STREQUAL "static")
    set(getter_args GETTER "${getter}")
endif()
set(runtime_args)
if(runtimeLibraries)
    set(runtime_args RUNTIME_LIBRARIES ${runtimeLibraries})
endif()
wio_add_vm_module(vm_module ${module_kind} BYTECODE "${bytecode}" ${getter_args}
    NATIVE_LIBRARIES ${producers} ${runtime_args})
output_path(vm_module "${module}")
if(host)
    if(NOT hostSources)
        message(FATAL_ERROR "VM project host has no sources")
    endif()
    add_executable(vm_host ${hostSources})
    target_compile_features(vm_host PRIVATE cxx_std_20)
    target_include_directories(vm_host PRIVATE ${hostIncludes})
    target_link_directories(vm_host PRIVATE ${hostLinkDirs})
    target_compile_options(vm_host PRIVATE ${hostArguments})
    target_link_options(vm_host PRIVATE ${hostLinkArguments})
    target_link_libraries(vm_host PRIVATE Wio::wio_sdk Threads::Threads ${CMAKE_DL_LIBS} ${hostLibraries})
    if(MSVC AND WioVmRuntime_MSVC_STATIC_RUNTIME)
        set_property(TARGET vm_host PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    output_path(vm_host "${host}")
    if(kind STREQUAL "static")
        target_link_libraries(vm_host PRIVATE vm_module)
        wio_deploy_vm_module(vm_module HOST vm_host DEPLOY_TARGET vm_repair)
        wio_install_vm_module(vm_module HOST vm_host DESTINATION bin COMPONENT WioVmProject)
    else()
        add_dependencies(vm_host vm_module)
        wio_deploy_vm_module(vm_module HOST vm_module DEPLOY_TARGET vm_repair)
        wio_install_vm_module(vm_module DESTINATION lib COMPONENT WioVmProject)
        install(TARGETS vm_host RUNTIME DESTINATION bin COMPONENT WioVmProject)
    endif()
elseif(kind STREQUAL "shared")
    wio_deploy_vm_module(vm_module HOST vm_module DEPLOY_TARGET vm_repair)
    wio_install_vm_module(vm_module DESTINATION lib COMPONENT WioVmProject)
else()
    # A standalone static archive is a link input, not an executable image.
    # Keep its bundle beside it for distribution; a consumer must deploy that
    # same relative bundle beside its final executable/shared image.
    get_target_property(stage vm_module WIO_VM_NATIVE_STAGE)
    get_target_property(manifest vm_module WIO_VM_NATIVE_MANIFEST)
    get_filename_component(directory "${module}" DIRECTORY)
    add_custom_target(vm_repair
        COMMAND "${CMAKE_COMMAND}" "-DMANIFEST=${manifest}" "-DSTAGE=${stage}"
            "-DDESTINATION=${directory}" -P "${package_dir}/DeployWioVmNative.cmake"
        DEPENDS vm_module VERBATIM)
    _wio_vm_literal(manifest_literal "${manifest}")
    _wio_vm_literal(stage_literal "${stage}")
    _wio_vm_literal(deploy_literal "${package_dir}/DeployWioVmNative.cmake")
    _wio_vm_literal(install_literal "${package_dir}/InstallWioVmNative.cmake")
    install(CODE "set(WIO_VM_INSTALL_MANIFEST ${manifest_literal})
set(WIO_VM_INSTALL_STAGE ${stage_literal})
set(WIO_VM_INSTALL_DESTINATION lib)
set(WIO_VM_INSTALL_DEPLOY ${deploy_literal})
include(${install_literal})" COMPONENT WioVmProject)
    install(TARGETS vm_module ARCHIVE DESTINATION lib COMPONENT WioVmProject)
endif()
get_target_property(manifest vm_module WIO_VM_NATIVE_MANIFEST)
get_target_property(stage vm_module WIO_VM_NATIVE_STAGE)
if(host)
    set(ready_dependency vm_host)
else()
    set(ready_dependency vm_module)
endif()
add_custom_target(wio_vm_project_ready
    COMMAND "${CMAKE_COMMAND}" -DACTION=record "-DMETADATA=${CMAKE_BINARY_DIR}/artifacts-$<CONFIG>.json"
        "-DEXPECTED_KIND=${kind}" "-DEXPECTED_MODULE=${module}" "-DEXPECTED_HOST=${host}"
        "-DNATIVE_MANIFEST=${manifest}" "-DNATIVE_STAGE=${stage}"
        -P "${package_dir}/ManageWioVmProject.cmake"
    DEPENDS ${ready_dependency} vm_repair VERBATIM)
